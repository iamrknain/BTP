% Slide 3: SC Mapping Revisited
\begin{frame}{Schwarz-Christoffel Mapping Revisited}
    \vfill
    \begin{center}
        \Large \[ w(z) = C_1 \int^z \prod_{k=1}^{n} (\zeta - x_k)^{-\alpha_k} \, d\zeta + C_2 \]
    \end{center}
    \vfill
    \begin{columns}[c]
        \column{0.6\textwidth}
        \begin{itemize}
            \item \textbf{\color{brightblue}Angle Parameters}: $\alpha_k = 1 - \theta_k/\pi$, interior angle $\theta_k$.
            \item \textbf{\color{brightblue}Boundary Correspondence}: Real axis maps to polygon boundary.
            \item \textbf{\color{brightblue}Argument Change}: $\arg(dw/dz)$ shifts by $\alpha_k \pi$ at each $x_k$.
        \end{itemize}
        \column{0.4\textwidth}
        \begin{center}

        \end{center}
    \end{columns}

% Slide 4: The Numerical Parameter Problem (3-Column Style)
\begin{frame}{The Numerical Parameter Problem}
    \vfill
    \begin{columns}[c]
        \column{0.3\textwidth}
        \textbf{\color{brightblue}The Core Issue} \\
        \footnotesize Side lengths depend on prevertex spacing $x_k$ on the real axis.
        \vfill
        \textbf{\color{brightblue}Degrees of Freedom} \\
        \footnotesize $n-3$ essential unknowns after normalization.
        
        \column{0.4\textwidth}
        \begin{premiumbox}{Residual Equations}
            \centering $F_j(X) = I_j(X) - \lambda_j I_1(X) = 0$
        \end{premiumbox}
        
        \column{0.3\textwidth}
        \textbf{\color{brightblue}Möbius Normalization} \\
        \footnotesize Three prevertices fixed arbitrarily (e.g., $-1, 0, 1$).
        \vfill
        \textbf{\color{brightblue}Geometric Constraint} \\
        \footnotesize Ordering $x_1 < x_2 < \dots < x_n$.
    \end{columns}
\end{frame}

% Slide 5: Newton Iteration
\begin{frame}{Solving the Parameter Problem: Newton Iteration}
    \vfill
    \begin{center}
        \Large \[ x^{(m+1)} = x^{(m)} + \eta h^{(m)}, \quad J(x^{(m)})h^{(m)} = -F(x^{(m)}) \]
    \end{center}
    \vfill
    \begin{itemize}
        \item \textbf{Linearization}: Jacobian $J_{j\nu}$ computed for each step.
        \item \textbf{Stopping Criteria}: Residual norm $\|F\| < 10^{-8}$.
        \item \textbf{Damping Factor}: $0 < \eta \le 1$ to preserve prevertex ordering.
    \end{itemize}
\end{frame}

% Slide 8: Bergman Kernel
\begin{frame}{The Bergman Kernel Method}
    \vfill
    \begin{premiumbox}{Reproducing Property}
        $\iint_D K(z, \zeta) f(\zeta) \, dS_\zeta = f(z)$
    \end{premiumbox}
    \vfill
    \begin{itemize}
        \item \textbf{Orthonormal Basis}: $K(z, \zeta) = \sum_{j} \phi_j(z) \overline{\phi_j(\zeta)}$.
        \item \textbf{Recovery}: $f(z) = \frac{1}{\sqrt{K(a,a)}} \int_a^z K(\zeta, a) \, d\zeta$.
    \end{itemize}
\end{frame}
